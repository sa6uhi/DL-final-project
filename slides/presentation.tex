\begin{frame}{Three gaps in fraud engines, three answers}
  \begin{columns}[T]
    \column{0.31\textwidth}
      \gapcard{1\enspace Isolated rows}
        {Scoring each transaction alone misses velocity bursts and behaviour shifts.}
        {FT-CAT Transformer}
        {Feature tokens cross-attend to the card's last $K{=}5$ transactions.}
    \column{0.31\textwidth}
      \gapcard{2\enspace Zero-day fraud}
        {Supervised models only recognise fraud seen in past labels.}
        {Denoising autoencoder}
        {Trained on legitimate traffic only; flags what does not reconstruct.}
    \column{0.31\textwidth}
      \gapcard{3\enspace Uncalibrated scores}
        {A raw score says nothing about when it can be trusted.}
        {Split conformal triage}
        {Prediction sets with a coverage target decide approve / block / review.}
  \end{columns}
  \vfill
  \keyline{Plus a \textbf{learned gate} fusing both detectors, shipped as a
    \textbf{FastAPI + Streamlit} product on CPU.}
\end{frame}

      \begin{beamercolorbox}[wd=\linewidth,sep=0.4em,rounded=true]{takeaway}
        \centering\small
        $S_{\mathrm{anomaly}}(x) = \sum_j (x_j-\hat{x}_j)^2 + \gamma_1 \sum_j |x_j-\hat{x}_j|$
      \end{beamercolorbox}
    \column{0.39\textwidth}
      \small\textbf{\color{brand}Input}\par
      \footnotesize 800-d vector: 400 scaled numeric features + 400 missingness flags.\par
      \vspace{0.55em}
      \small\textbf{\color{brand}Training}\par
      \footnotesize Legitimate transactions only ($y{=}0$), with Gaussian noise ($\sigma{=}0.1$)
      and 10\% feature dropout, so it learns structure instead of copying.\par
      \vspace{0.55em}
      \small\textbf{\color{brand}Output}\par
      \footnotesize Residual score with $\gamma_1{=}0.4$, normalised by its 99.9th
      training percentile; the $L_1$ term keeps a few large residuals from dominating.\par
      \vspace{0.55em}
      \footnotesize\color{muted}485,440 parameters\enspace\textperiodcentered\enspace 1.95 MB
  \end{columns}
\end{frame}

\begin{frame}{Limitations we report, and what we learned}
  \vspace{0.5em}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.55em,rounded=true]{card}
        \small\textbf{\color{block}Limitations}\par\vspace{0.3em}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.35em}
          \item \textbf{Class asymmetry:} marginal coverage 98.99\%, but fraud-class
                coverage is only 70.9\%.
          \item \textbf{No auto-blocking} at $\alpha{=}0.01$: it is an approve-or-review system.
          \item \textbf{Temporal drift} breaks exchangeability, so 98.99\% is measured,
                not guaranteed.
          \item \textbf{Explanations} cover the DAE component, not the fused decision.
          \item \textbf{History key} card1 may partly encode card identity.
        \end{itemize}
      \end{beamercolorbox}
    \column{0.48\textwidth}
      \begin{beamercolorbox}[wd=\linewidth,sep=0.55em,rounded=true]{card}
        \small\textbf{\color{brand}Lessons}\par\vspace{0.3em}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{0.35em}
          \item \textbf{Evaluate out of time:} PR-AUC fell from 0.44 in development
                to 0.34 on the final test period.
          \item \textbf{Complexity must earn its place:} a flat MLP matched or beat
                FT-CAT, and trees still lead.
          \item \textbf{Isolate test side effects:} a test once overwrote the serving
                threshold artifact. It was caught, fixed and regenerated.
          \item \textbf{Report the operating point}, not just a score:
                coverage and workload trade off.
        \end{itemize}
      \end{beamercolorbox}
  \end{columns}
\end{frame}

\end{frame}
